3use glib::{Value, translate::*};
4
5use crate::{TreeIter, TreeModel, TreeModelFilter, ffi, prelude::*, subclass::prelude::*};
6
7pub trait TreeModelFilterImpl: ObjectImpl + ObjectSubclass<Type: IsA<TreeModelFilter>> {
8    fn visible(&self, child_model: &TreeModel, child_iter: &TreeIter) -> bool {
9        self.parent_visible(child_model, child_iter)
10    }
11
12    fn modify(&self, child_model: &TreeModel, iter: &TreeIter, column: i32) -> Value {
13        self.parent_modify(child_model, iter, column)
14    }
15}
16
17pub trait TreeModelFilterImplExt: TreeModelFilterImpl {
18    // Whether the row indicated by iter is visible
19    fn parent_visible(&self, child_model: &TreeModel, child_iter: &TreeIter) -> bool {
20        unsafe {
21            let data = Self::type_data();
22            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkTreeModelFilterClass;
23            if let Some(f) = (*parent_class).visible {
24                from_glib(f(
25                    self.obj()
26                        .unsafe_cast_ref::<TreeModelFilter>()
27                        .to_glib_none()
28                        .0,
29                    child_model.to_glib_none().0,
30                    mut_override(child_iter.to_glib_none().0),
31                ))
32            } else {
33                true // always visible if not set
34            }
35        }
36    }
37
38    fn parent_modify(&self, child_model: &TreeModel, iter: &TreeIter, column: i32) -> Value {
39        unsafe {
40            let data = Self::type_data();
41            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkTreeModelFilterClass;
42            let f = (*parent_class)
43                .modify
44                .expect("No parent class impl for \"modify\"");
45            let mut value = Value::uninitialized();
46            f(
47                self.obj()
48                    .unsafe_cast_ref::<TreeModelFilter>()
49                    .to_glib_none()
50                    .0,
51                child_model.to_glib_none().0,
52                mut_override(iter.to_glib_none().0),
53                value.to_glib_none_mut().0,
54                column,
55            );
56            value
57        }
58    }
59}
60
61impl<T: TreeModelFilterImpl> TreeModelFilterImplExt for T {}
62
63unsafe impl<T: TreeModelFilterImpl> IsSubclassable<T> for TreeModelFilter {
64    fn class_init(class: &mut glib::Class<Self>) {
65        Self::parent_class_init::<T>(class);
66
67        if !crate::rt::is_initialized() {
68            panic!("GTK has to be initialized first");
69        }
70
71        let klass = class.as_mut();
72        klass.visible = Some(tree_model_filter_visible::<T>);
73        klass.modify = Some(tree_model_filter_modify::<T>);
74    }
75}

77unsafe extern "C" fn tree_model_filter_visible<T: TreeModelFilterImpl>(
78    ptr: *mut ffi::GtkTreeModelFilter,
79    child_modelptr: *mut ffi::GtkTreeModel,
80    iterptr: *mut ffi::GtkTreeIter,
81) -> glib::ffi::gboolean {
82    unsafe {
83        let instance = &*(ptr as *mut T::Instance);
84        let imp = instance.imp();
85        let child_model: Borrowed<TreeModel> = from_glib_borrow(child_modelptr);
86        let iter: Borrowed<TreeIter> = from_glib_borrow(iterptr);
87
88        imp.visible(&child_model, &iter).into_glib()
89    }
90}
91
92unsafe extern "C" fn tree_model_filter_modify<T: TreeModelFilterImpl>(
93    ptr: *mut ffi::GtkTreeModelFilter,
94    child_modelptr: *mut ffi::GtkTreeModel,
95    iterptr: *mut ffi::GtkTreeIter,
96    valueptr: *mut glib::gobject_ffi::GValue,
97    column: i32,
98) {
99    unsafe {
100        let instance = &*(ptr as *mut T::Instance);
101        let imp = instance.imp();
102        let child_model: Borrowed<TreeModel> = from_glib_borrow(child_modelptr);
103        let iter: Borrowed<TreeIter> = from_glib_borrow(iterptr);
104
105        let v = imp.modify(&child_model, &iter, column);
106
107        // `valueptr` has not been initialized, so no need to unset first.  Then consume `v` and
108        // transfer ownership of its bits to `valueptr`.
109        std::ptr::write(valueptr, v.into_raw());
110    }
111}

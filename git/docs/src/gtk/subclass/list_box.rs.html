15pub trait ListBoxImpl: ListBoxImplExt + ContainerImpl + WidgetImpl {
16    /// Class handler for the [`activate-cursor-row`][struct@crate::ListBox#activate-cursor-row] signal
17    fn activate_cursor_row(&self) {
18        self.list_box_activate_cursor_row()
19    }
20
21    /// Class handler for the [`move-cursor`][struct@crate::ListBox#move-cursor] signal
22    fn move_cursor(&self, step: MovementStep, count: i32) {
23        self.list_box_move_cursor(step, count)
24    }
25
26    /// Class handler for the [`row-activated`][struct@crate::ListBox#row-activated] signal
27    fn row_activated(&self, row: &ListBoxRow) {
28        self.list_box_row_activated(row)
29    }
30
31    /// Class handler for the [`row-selected`][struct@crate::ListBox#row-selected] signal
32    fn row_selected(&self, row: Option<&ListBoxRow>) {
33        self.list_box_row_selected(row)
34    }
35
36    /// Select all children of `box_`, if the selection mode allows it.
37    fn select_all(&self) {
38        self.list_box_select_all()
39    }
40
41    /// Class handler for the [`selected-rows-changed`][struct@crate::ListBox#selected-rows-changed] signal
42    fn selected_rows_changed(&self) {
43        self.list_box_selected_rows_changed()
44    }
45
46    /// Class handler for the [`toggle-cursor-row`][struct@crate::ListBox#toggle-cursor-row] signal
47    fn toggle_cursor_row(&self) {
48        self.list_box_toggle_cursor_row()
49    }
50
51    /// Unselect all children of `box_`, if the selection mode allows it.
52    fn unselect_all(&self) {
53        self.list_box_unselect_all()
54    }
55}

62pub trait ListBoxImplExt: ObjectSubclass + sealed::Sealed {
63    fn list_box_activate_cursor_row(&self) {
64        unsafe {
65            let data = Self::type_data();
66            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
67            if let Some(f) = (*parent_class).activate_cursor_row {
68                f(self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0)
69            }
70        }
71    }
72    fn list_box_move_cursor(&self, step: MovementStep, count: i32) {
73        unsafe {
74            let data = Self::type_data();
75            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
76            if let Some(f) = (*parent_class).move_cursor {
77                f(
78                    self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0,
79                    step.into_glib(),
80                    count,
81                );
82            }
83        }
84    }
85    fn list_box_row_activated(&self, row: &ListBoxRow) {
86        unsafe {
87            let data = Self::type_data();
88            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
89            if let Some(f) = (*parent_class).row_activated {
90                f(
91                    self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0,
92                    row.to_glib_none().0,
93                )
94            }
95        }
96    }
97    fn list_box_row_selected(&self, row: Option<&ListBoxRow>) {
98        unsafe {
99            let data = Self::type_data();
100            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
101            if let Some(f) = (*parent_class).row_selected {
102                f(
103                    self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0,
104                    mut_override(row.to_glib_none().0),
105                )
106            }
107        }
108    }
109    fn list_box_select_all(&self) {
110        unsafe {
111            let data = Self::type_data();
112            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
113            if let Some(f) = (*parent_class).select_all {
114                f(self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0)
115            }
116        }
117    }
118    fn list_box_selected_rows_changed(&self) {
119        unsafe {
120            let data = Self::type_data();
121            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
122            if let Some(f) = (*parent_class).selected_rows_changed {
123                f(self.obj().unsafe_cast_ref::<ListBox>().to_glib_none().0)
124            }
125        }
126    }
127    fn list_box_toggle_cursor_row(&self) {
128        unsafe {
129            let data = Self::type_data();
130            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;
131            if let Some(f) = (*parent_class).toggle_cursor_row {

136    fn list_box_unselect_all(&self) {

138            let data = Self::type_data();
139            let parent_class = data.as_ref().parent_class() as *mut ffi::GtkListBoxClass;

149unsafe impl<T: ListBoxImpl> IsSubclassable<T> for ListBox {
150    fn class_init(class: &mut ::glib::Class<Self>) {
151        Self::parent_class_init::<T>(class);
152
153        if !crate::rt::is_initialized() {
154            panic!("GTK has to be initialized first");
155        }
156
157        let klass = class.as_mut();
158        klass.activate_cursor_row = Some(list_box_activate_cursor_row::<T>);
159        klass.move_cursor = Some(list_box_move_cursor::<T>);
160        klass.row_activated = Some(list_box_row_activated::<T>);
161        klass.row_selected = Some(list_box_row_selected::<T>);
162        klass.select_all = Some(list_box_select_all::<T>);
163        klass.selected_rows_changed = Some(list_box_selected_rows_changed::<T>);
164        klass.toggle_cursor_row = Some(list_box_toggle_cursor_row::<T>);
165        klass.unselect_all = Some(list_box_unselect_all::<T>);
166    }
167}

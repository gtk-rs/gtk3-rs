651    /// Returns the clipboard object for the given selection.
652    /// Cut/copy/paste menu items and keyboard shortcuts should use
653    /// the default clipboard, returned by passing `GDK_SELECTION_CLIPBOARD` for `selection`.
654    /// (`GDK_NONE` is supported as a synonym for GDK_SELECTION_CLIPBOARD
655    /// for backwards compatibility reasons.)
656    /// The currently-selected object or text should be provided on the clipboard
657    /// identified by `GDK_SELECTION_PRIMARY`. Cut/copy/paste menu items
658    /// conceptually copy the contents of the `GDK_SELECTION_PRIMARY` clipboard
659    /// to the default clipboard, i.e. they copy the selection to what the
660    /// user sees as the clipboard.
661    ///
662    /// (Passing `GDK_NONE` is the same as using `gdk_atom_intern
663    /// ("CLIPBOARD", FALSE)`.
664    ///
665    /// See the
666    /// [FreeDesktop Clipboard Specification](http://www.freedesktop.org/Standards/clipboards-spec)
667    /// for a detailed discussion of the “CLIPBOARD” vs. “PRIMARY”
668    /// selections under the X window system. On Win32 the
669    /// `GDK_SELECTION_PRIMARY` clipboard is essentially ignored.)
670    ///
671    /// It’s possible to have arbitrary named clipboards; if you do invent
672    /// new clipboards, you should prefix the selection name with an
673    /// underscore (because the ICCCM requires that nonstandard atoms are
674    /// underscore-prefixed), and namespace it as well. For example,
675    /// if your application called “Foo” has a special-purpose
676    /// clipboard, you might call it “_FOO_SPECIAL_CLIPBOARD”.
677    /// ## `display`
678    /// the [`gdk::Display`][crate::gdk::Display] for which the clipboard is to be retrieved or created.
679    /// ## `selection`
680    /// a [`gdk::Atom`][crate::gdk::Atom] which identifies the clipboard to use.
681    ///
682    /// # Returns
683    ///
684    /// the appropriate clipboard object. If no
685    ///  clipboard already exists, a new one will be created. Once a clipboard
686    ///  object has been created, it is persistent and, since it is owned by
687    ///  GTK+, must not be freed or unrefd.
688    #[doc(alias = "gtk_clipboard_get_for_display")]
689    #[doc(alias = "get_for_display")]
690    pub fn for_display(display: &gdk::Display, selection: &gdk::Atom) -> Clipboard {
691        assert_initialized_main_thread!();
692        unsafe {
693            from_glib_none(ffi::gtk_clipboard_get_for_display(
694                display.to_glib_none().0,
695                selection.to_glib_none().0,
696            ))
697        }
698    }
